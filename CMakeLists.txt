cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_LIST_DIR}/project.json" JESTER_PROJECT_JSON)
string(JSON JESTER_VERSION GET "${JESTER_PROJECT_JSON}" version)
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" JESTER_VERSION_NUMERIC "${JESTER_VERSION}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/project.json")
if(NOT PROJECT_NAME)
    set(JESTER_O3DE_ROOT "D:/O3DE" CACHE PATH "Installed O3DE SDK")
    list(PREPEND CMAKE_MODULE_PATH "${JESTER_O3DE_ROOT}/cmake")
    find_package(o3de REQUIRED)
    project(Jester VERSION ${JESTER_VERSION_NUMERIC} LANGUAGES C CXX)
    o3de_initialize()
endif()
